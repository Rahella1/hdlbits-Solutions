module top_module(
    input clk,
    input load,
    input [255:0] data,
    output reg [255:0] q
);

    // 2D representation for easier coordinate indexing: grid[row][col]
    wire [15:0] grid [15:0];
    reg  [255:0] q_next;

    // Map 1D vector q[255:0] into 2D array grid[15:0][15:0]
    genvar r, c;
    generate
        for (r = 0; r < 16; r = r + 1) begin : pack_grid
            for (c = 0; c < 16; c = c + 1) begin : pack_col
                assign grid[r][c] = q[r*16 + c];
            end
        end
    endgenerate

    // Compute next state for all 256 cells in parallel
    integer i, j;
    integer count;
    integer r_up, r_down, c_left, c_right;

    always @(*) begin
        for (i = 0; i < 16; i = i + 1) begin
            for (j = 0; j < 16; j = j + 1) begin
                
                // Toroidal wrap-around indices (modulo 16)
                r_up    = (i == 15) ? 0  : (i + 1);
                r_down  = (i == 0)  ? 15 : (i - 1);
                c_right = (j == 15) ? 0  : (j + 1);
                c_left  = (j == 0)  ? 15 : (j - 1);

                // Sum all 8 neighbors
                count = grid[r_up][c_left]   + grid[r_up][j]   + grid[r_up][c_right] +
                        grid[i][c_left]                        + grid[i][c_right]    +
                        grid[r_down][c_left] + grid[r_down][j] + grid[r_down][c_right];

                // Conway's Game of Life state rules
                case (count)
                    2:       q_next[i*16 + j] = grid[i][j]; // State unchanged
                    3:       q_next[i*16 + j] = 1'b1;       // Cell becomes alive
                    default: q_next[i*16 + j] = 1'b0;       // Dies (underpopulation or overpopulation)
                endcase
            end
        end
    end

    // Sequential update
    always @(posedge clk) begin
        if (load) begin
            q <= data;
        end else begin
            q <= q_next;
        end
    end

endmodule
